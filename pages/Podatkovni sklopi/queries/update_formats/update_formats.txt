INSERT INTO metadata.dataset_formats (dataset_id, format_id, sort_order)
SELECT 
  CAST({{ JSONForm1.formData.id || appsmith.store.viewData.id || Tabela_sklopov.triggeredRow.id || Tabela_sklopov.selectedRow.id }} AS INTEGER) AS dataset_id,
  CAST(format_id AS INTEGER) AS format_id,
  ordinality AS sort_order
FROM jsonb_array_elements_text(
  -- Fallback to empty JSON array if null/empty string
  CASE 
    WHEN {{ JSON.stringify(JSONForm1.formData.data_formats || JSONForm1.formData.formats || []) }} = '' 
      THEN '[]'::jsonb
    ELSE {{ JSON.stringify(JSONForm1.formData.data_formats || JSONForm1.formData.formats || []) }}::jsonb
  END
) WITH ORDINALITY AS t(format_id, ordinality);
