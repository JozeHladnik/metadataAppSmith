SELECT 
    dls.id, 
    l.label AS Location, 
    stg.label AS Stage
FROM metadata.dataset_locations_stage dls
JOIN metadata.locations l ON dls.location_id = l.id
LEFT JOIN metadata.stages stg ON dls.stage_id = stg.id
WHERE dls.dataset_id = {{ Tabela_sklopov.triggeredRow.id }};
