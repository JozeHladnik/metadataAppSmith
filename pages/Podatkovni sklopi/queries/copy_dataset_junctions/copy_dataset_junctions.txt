SELECT
  CAST({{ this.params.newId }} AS INTEGER),
  author_id,
  COALESCE(sort_order, 0) + 1,
  is_primary
FROM metadata.dataset_author
WHERE dataset_id = CAST({{ this.params.oldId }} AS INTEGER);
